cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(batch_max_sum_correctness LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
if(NOT DEFINED NPU_ARCH)
    set(NPU_ARCH dav-2201)
endif()
add_executable(correctness main.asc)
target_link_libraries(correctness PRIVATE tiling_api register platform unified_dlog dl m graph_base)
target_compile_options(correctness PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)

option(BUILD_SIMULATOR_BRIDGE "构建仿真专用动态加载入口" OFF)
if(BUILD_SIMULATOR_BRIDGE)
    add_library(simulator_bridge SHARED simulator_bridge.asc)
    target_link_libraries(simulator_bridge PRIVATE tiling_api register platform unified_dlog dl m graph_base)
    target_compile_options(simulator_bridge PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)
endif()
